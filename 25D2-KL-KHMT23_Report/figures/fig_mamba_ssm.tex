\begin{tikzpicture}
  \node[io] (xt) {$x_t$};
  \node[blk, right=12mm of xt] (bt) {$b_t\odot$};
  \node[op, right=12mm of bt] (plus) {$+$};
  \node[state, right=12mm of plus] (h) {$h_t$};
  \node[blk, right=12mm of h] (ct) {$\odot\, c_t$};
  \node[io, right=12mm of ct] (yt) {$y_t$};

  \draw[flow] (xt) -- (bt);
  \draw[flow] (bt) -- (plus);
  \draw[flow] (plus) -- (h);

% vòng hồi quy: h_{t-1} nhân a rồi cộng vào
  \node[blk, below=9mm of plus, minimum width=20mm] (a) {$a \odot h_{t-1}$};
  \draw[flow] (h.south) |- (a.east);
  \draw[flow] (a.north) -- (plus.south);
  \node[lbl, below=1mm of a] {$a=e^{-\mathrm{softplus}(\log a)}\in(0,1)$};

  % tính chọn lọc: b_t, c_t, (Delta) phụ thuộc đầu vào x_t
  \node[blk, above=9mm of h, minimum width=44mm, draw=cvblue, fill=cvbluefill] (sel)

  \draw[sel] (xt.north) |- (sel.west);
  \draw[sel] (sel.south) -- (bt.north);
  \draw[sel] (sel.south) -- (ct.north);

  \node[lbl, below=6mm of a, text width=11cm, align=center]
     {Hồi quy: $h_t = a\odot h_{t-1} + b_t\odot x_t$,\quad $y_t = h_t\odot c_t$ \; --- độ phức tạp tuyến tính theo chiều dài chuỗi.};
